% UML actor (stick figure) with a label below: \umlactor{name}{x,y}{label}
\newcommand{\umlactor}[3]{%
  \begin{scope}[shift={(#2)}]
    \draw (0,0.55) circle (0.18);
    \draw (0,0.37) -- (0,-0.1);
    \draw (-0.3,0.2) -- (0.3,0.2);
    \draw (0,-0.1) -- (-0.22,-0.5);
    \draw (0,-0.1) -- (0.22,-0.5);
    \node[font=\footnotesize, align=center, anchor=north] (#1) at (0,-0.6) {#3};
    \coordinate (#1-c) at (0,0.1);
  \end{scope}}

% Sequence diagram helpers.
% \seqlife{name}{x}{label}{bottom y}: lifeline head at y=0 and dashed line down
\newcommand{\seqlife}[4]{%
  \node[lifehead] (#1) at (#2,0) {#3};
  \draw[lifeline] (#1.south) -- (#2,#4);}
% \seqmsg{x from}{x to}{y}{label} and \seqret{...} for dashed replies
\newcommand{\seqmsg}[4]{\draw[msg] (#1,#3) -- node[msglabel]{#4} (#2,#3);}
\newcommand{\seqret}[4]{\draw[reply] (#1,#3) -- node[msglabel]{#4} (#2,#3);}
% \seqnote{x}{y}{text}: note box whose top-centre sits at (x,y)
\newcommand{\seqnote}[3]{\node[note, anchor=north] at (#1,#2) {#3};}
\newcommand{\ssl}{\textless}
\newcommand{\ssr}{\textgreater}

% UML component glyph in the top-right corner of a node
\newcommand{\compglyph}[1]{%
  \draw[fill=white] ($(#1.north east)+(-0.42,-0.14)$) rectangle ++(0.26,-0.32);
  \draw[fill=white] ($(#1.north east)+(-0.48,-0.2)$) rectangle ++(0.12,-0.06);
  \draw[fill=white] ($(#1.north east)+(-0.48,-0.32)$) rectangle ++(0.12,-0.06);}
